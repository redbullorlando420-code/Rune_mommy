        if mode == 'play':
            tick_crowd(self, dt)
            tick_traffic(self, dt)
            self._tick_heat(dt)
            self._tick_tutorial(dt)
        # safe: skip crowd/traffic/heat; walk only

        if self.ui_open:
            self._refresh_hud()
            return

        if mode in ('play', 'safe') and self.in_car:
            self._drive(dt)
        else:
            self._walk(dt)

        if mode == 'play':
            self._maybe_auto_michelle()
            self._tick_visit_quests()
            if self.pistol_drawn and self.owned_pistol and held_keys['left mouse'] and self.shoot_cd <= 0:
                self._shoot()

        self._refresh_hud()

    def _walk(self, dt):
        from ursina import held_keys, mouse, raycast, Vec3
        player = self.player
        mode = getattr(self, 'mode', 'play')

        # Look only while the relative cursor is captured. Moving a free Tab
        # cursor must not rotate/invert the view behind the player's back.
        sens = float(getattr(self, 'mouse_sens', 140) or 140)
        skip_look = int(getattr(self, 'mouse_look_cooldown', 0) or 0)
        # Use game state rather than Panda's transient locked flag. Some
        # Windows/Panda transitions report unlocked for a frame after leaving
        # the safe yard even though the game owns mouse-look.
        if not skip_look and not getattr(self, 'mouse_free', False) and not self.ui_open:
            player.rotation_y += mouse.velocity[0] * sens
            # Positive mouse Y looks upward, matching conventional desktop controls.
            self.cam_pivot.rotation_x += mouse.velocity[1] * sens
            self.cam_pivot.rotation_x = max(-35, min(40, self.cam_pivot.rotation_x))
        # Q/E and arrows turn even if mouse is dead
        if held_keys['q'] or held_keys['left arrow']:
            player.rotation_y += 90 * dt
        if held_keys['e'] and mode == 'safe':
            # E is gate in safe — only turn with E if not near gate; use right arrow instead
            pass
        if held_keys['right arrow']:
            player.rotation_y -= 90 * dt

        if mode == 'play' and not self.ui_open and not getattr(self, 'mouse_free', False):
            self._lock_mouse(True)

        down = lambda key: self._movement_down(held_keys, key)
        wish_x = int(down('d')) - int(down('a'))
        wish_z = int(down('w')) - int(down('s'))
        # Arrow keys as backup movement
        wish_x += held_keys['right arrow'] * 0  # right arrow is turn
        wish_z += 0
        if down('up arrow'):
            wish_z += 1
        if down('down arrow'):
            wish_z -= 1
        wish = Vec3(wish_x, 0, wish_z)
        if wish.length() > 0:
            wish = wish.normalized()
            speed = 10.0 if mode == 'safe' else 8.5
            move = (player.forward * wish.z + player.right * wish.x) * speed * dt
            if mode == 'safe':
                # Nuclear: no clip tests in the yard — just move and clamp.
                player.position += move
                cx, _cy, cz = getattr(self, 'safe_yard_center', (80.0, 0.0, 80.0))
                player.x = max(cx - 12.0, min(cx + 12.0, player.x))
                player.z = max(cz - 12.0, min(cz + 12.0, player.z))
            else:
                origin = player.world_position + Vec3(0, 0.9, 0)
                ign = list(self.ignore)
                hit = raycast(origin, move.normalized(), distance=0.55 + move.length(), ignore=ign, debug=False)
                if not hit.hit:
                    player.position += move
                else:
                    mx = Vec3(move.x, 0, 0)
                    mz = Vec3(0, 0, move.z)
                    if mx.length() > 1e-5:
                        hx = raycast(origin, mx.normalized(), distance=0.55 + mx.length(), ignore=ign, debug=False)
                        if not hx.hit:
                            player.x += mx.x
                    if mz.length() > 1e-5:
                        hz = raycast(origin, mz.normalized(), distance=0.55 + mz.length(), ignore=ign, debug=False)
                        if not hz.hit:
                            player.z += mz.z
                player.x = max(-330, min(330, player.x))
                player.z = max(-165, min(195, player.z))

        self.y_vel -= 32 * dt
        player.y += self.y_vel * dt
        if player.y <= 0:
            player.y = 0
            self.y_vel = 0
            self.grounded = True
        else:
            self.grounded = False

        if held_keys['space'] and self.grounded:
            self.y_vel = 8.2
            self.grounded = False

        if not self.visual.enabled:
            self.visual.enabled = True
        if self.cam_in_car:
            self._set_camera_follow(False)
        self._update_camera_collision()


    def _drive(self, dt):
        from ursina import held_keys, mouse, raycast, Vec3, Entity, destroy, color
        from vendor.vehicle import bicycle_step
        car = self.in_car
        if not getattr(self, 'mouse_free', False) and not int(getattr(self, 'mouse_look_cooldown', 0) or 0):
            # Clamp mouse look while driving (pitch only on cam_pivot)
            car.cam_pivot.rotation_x += mouse.velocity[1] * (float(getattr(self, "mouse_sens", 140) or 140) * 0.57)
            car.cam_pivot.rotation_x = max(-20, min(25, car.cam_pivot.rotation_x))

        dmg = float(getattr(car, 'damage', 0.0) or 0.0)
        max_damage = float(getattr(car, 'max_damage', 240.0) or 240.0)
        dmg_mul = max(0.30, 1.0 - 0.48 * (dmg / max_damage))
        base_vmax, base_accel = 24.0, 16.0
        vmax = base_vmax * dmg_mul
        accel_force = base_accel * dmg_mul
        brake_force = 0.0
        accel = 0.0

        fuel = float(getattr(car, 'fuel', 100.0) or 0.0)
        out_of_gas = fuel <= 0.0 or getattr(car, 'disabled', False)

        if out_of_gas:
            # Speed decays to 0; no throttle / reverse
            brake_force = 14.0
            if not getattr(car, 'out_of_gas_toasted', False) and fuel <= 0.0 and not getattr(car, 'disabled', False):
                car.out_of_gas_toasted = True
                self.toast('Out of gas. Find a pump.')
        else:
            if held_keys['w']:
                accel = accel_force
            elif held_keys['s']:
                # S brakes if moving forward; hold for reverse when nearly stopped
                if car.speed > 0.55:
                    brake_force = 32.0
                else:
                    accel = -accel_force * 0.55
            if held_keys['space']:
                # Stronger handbrake
                brake_force = max(brake_force, 52.0)

        # Bicycle yaw uses positive = right (+X), therefore D must be positive.
        # Reduce lock at speed to prevent arcade-like instant direction changes.
        steer_input = float(held_keys['d'] - held_keys['a'])
        speed_steer = max(0.34, 1.0 - abs(float(getattr(car, 'speed', 0.0))) / max(1.0, vmax) * 0.56)
        steer = steer_input * speed_steer
        if brake_force and abs(car.speed) > 0.05:
            # Decelerate in the direction opposite current travel. The bicycle
            # helper has no separate brake-force parameter.
            accel = -math.copysign(brake_force, car.speed)
        # Higher grip: less drag accelerating, more coasting
        nx, nz, nyaw, nspd = bicycle_step(
            car.x, car.z, car.rotation_y, car.speed,
            steer, accel, dt,
            vmax=vmax, wheelbase=2.6,
            # The bundled vehicle helper accepts one drag value. Brake force
            # is applied above by reducing acceleration / speed, rather than
            # passing unsupported keyword arguments that crash the game.
            drag=3.2 if brake_force else 0.55,
        )

        # Fuel burn ~ full tank lasts a few minutes of continuous drive
        # k_speed=0.028, k_accel=0.045  → cruise ~3 min; floored ~1–1.5 min
        if not getattr(car, 'traffic', False) and not out_of_gas:
            burn = (0.028 * abs(nspd) + 0.045 * max(0.0, accel)) * dt
            fuel = max(0.0, fuel - burn)
            car.fuel = fuel
            if fuel <= 0.0 and not getattr(car, 'out_of_gas_toasted', False):
                car.out_of_gas_toasted = True
                self.toast('Out of gas. Find a pump.')

        # Collision: buildings via raycast (do not ignore other cars); traffic via proximity
        proposed = Vec3(nx - car.x, 0, nz - car.z)
        hit_something = False
        impact_speed = abs(nspd)
        if proposed.length() > 0.001:
            origin = car.world_position + Vec3(0, 0.7, 0)
            ign = [self.player, self.visual, self.cam_pivot, car, getattr(car, 'body', None)]
            if self.gun_model:
                ign.append(self.gun_model)
            # Never destroy / count named NPC talk entities as crash targets
            for npc in self.npcs:
                if not npc:
                    continue
                kind = getattr(npc, 'kind', '')
                named = getattr(npc, 'npc_id', '')
                if kind in ('mira', 'michelle', 'gun', 'talk', 'lila', 'rosa', 'yara') or named in (
                    'mira', 'michelle', 'gage', 'lila', 'rosa', 'yara', 'rita', 'deb',
                    'motel_clerk', 'barista', 'pharmacist', 'neighbor', 'lot_attendant',
                    'amanda_friend', 'club27_host', 'quiet_spa_tech', 'citrus_guide', 'park_ranger',
                ):
                    ign.append(npc)
            ign = [e for e in ign if e]
            direction = proposed.normalized()
            hit = raycast(origin, direction, distance=max(2.0, proposed.length() + 1.2), ignore=ign, debug=False)
            if hit.hit:
                hit_something = True

        if not hit_something:
            for other in self.cars:
                if not other or other is car:
                    continue
                d = dist_xz((nx, 0, nz), (other.x, 0, other.z))
                if d < 2.6:
                    hit_something = True
                    break

        if hit_something:
            self._car_take_damage(car, impact_speed)
            nspd *= -0.25
            # stay put on impact
            nx, nz = car.x, car.z
        else:
            car.x, car.z = nx, nz
            car.rotation_y = nyaw

        car.speed = nspd
        car.y = 0
        if getattr(self, 'mode', 'play') == 'safe':
            cx, _cy, cz = getattr(self, 'safe_yard_center', (80.0, 0.0, 80.0))
            car.x = max(cx - 11.4, min(cx + 11.4, car.x))
            car.z = max(cz - 11.4, min(cz + 11.4, car.z))
        else:
            car.x = max(-330, min(330, car.x))
            car.z = max(-165, min(195, car.z))

        self._update_car_damage_visual(car, dt)

        # Slight camera follow lag (lerp cam_pivot local z toward target)
        try:
            target_z = -0.15 if abs(car.speed) > 8 else 0.2
            cur = float(getattr(car.cam_pivot, 'z', 0.2) or 0.2)
            car.cam_pivot.z = cur + (target_z - cur) * min(1.0, 3.5 * dt)
        except Exception:
            pass

        self.player.position = car.position
        self.player.y = 0
        self.player.rotation_y = car.rotation_y
        if self.visual.enabled:
            self.visual.enabled = False
        if not self.cam_in_car:
            self._set_camera_follow(True)
        self._update_camera_collision()

    def _car_take_damage(self, car, impact_speed):
        if getattr(car, 'traffic', False) and self.in_car is not car:
            return
        max_damage = float(getattr(car, 'max_damage', 240.0) or 240.0)
        # A slow curb contact should scuff, not total a car. High-speed impacts
        # still matter, but the larger integrity budget supports long drives.
        add = min(28.0, max(1.0, abs(float(impact_speed)) * 1.15))
        car.damage = min(max_damage, float(getattr(car, 'damage', 0.0) or 0.0) + add)
        if car.damage >= max_damage and not getattr(car, 'disabled', False):
            car.disabled = True
            car.speed = 0.0
            self.toast('Car totaled.')
            if self.in_car is car:
                # stall — force exit shortly via disabled flag in enter check
                pass
