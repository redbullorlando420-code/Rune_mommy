        if door_xz and not self.michelle_seen:
            dd = dist_xz(pos, (door_xz[0], 0, door_xz[1]))
            if dd < 3.0:
                options.append((dd, 'E  Sanctuary Drive'))
        for poi in getattr(self, 'pois', []) or []:
            if poi.get('is_gas'):
                continue
            pd = dist_xz(pos, poi['pos'])
            if pd < 3.4:
                options.append((pd, f"E  {poi.get('name', 'POI')}"))
        if not options:
            return ''
        options.sort(key=lambda x: x[0])
        return options[0][1]

    def _actor_pos(self):
        if self.in_car:
            p = self.in_car.world_position
            return (p.x, p.y, p.z)
        p = self.player.world_position
        return (p.x, p.y, p.z)

    def _nearest_car(self, pos, radius):
        best, best_d = None, radius
        for car in self.cars:
            if not car:
                continue
            if getattr(car, 'disabled', False):
                continue
            d = dist_xz(pos, (car.x, 0, car.z))
            if d < best_d:
                best, best_d = car, d
        return best, best_d

    def _nearest_gas(self, pos, radius=None):
        best, best_d = None, 99.0
        for poi in getattr(self, 'pois', []) or []:
            if not poi.get('is_gas'):
                continue
            r = float(radius if radius is not None else poi.get('pump_r', 4.2))
            d = dist_xz(pos, poi['pos'])
            if d < r and d < best_d:
                best, best_d = poi, d
        return best

    def _nearest_npc(self, pos, radius):
        best, best_d = None, radius
        for npc in self.npcs:
            if not npc:
                continue
            kind = getattr(npc, 'kind', '')
            if kind in ('target', 'mob', 'civilian', 'lot_rat', 'walker', 'thug', 'heat_hunter', 'barrel'):
                continue
            if getattr(npc, 'hittable', False):
                continue
            if getattr(npc, 'enabled', True) is False:
                continue
            d = dist_xz(pos, (npc.x, 0, npc.z))
            if d < best_d:
                best, best_d = npc, d
        return best, best_d

    def _nearest_stall(self, pos, radius):
        best, best_d = None, radius
        for st in self.stalls:
            d = dist_xz(pos, st['pos'])
            if d < best_d:
                best, best_d = st, d
        return best, best_d

    # ----- input ----------------------------------------------------------

    def _record_movement_key(self, key):
        """Mirror key transitions independent of Ursina's GUI focus state."""
        released = key.endswith(' up')
        base = key[:-3] if released else key
        if base not in ('w', 'a', 's', 'd', 'up arrow', 'down arrow'):
            return
        keys = getattr(self, 'move_keys', None)
        if keys is None:
            keys = self.move_keys = set()
        if released:
            keys.discard(base)
        else:
            keys.add(base)

    def _movement_down(self, held_keys, key):
        """Use direct input when a menu click left held_keys stale."""
        try:
            if held_keys[key]:
                return True
        except Exception:
            pass
        return key in getattr(self, 'move_keys', set())

    def on_input(self, key):
        from ursina import application, mouse
        mode = getattr(self, 'mode', 'play')
        self._record_movement_key(key)

        if mode in ('menu', 'settings'):
            # Button focus must never carry a relative-mouse lock into a menu
            # or settings page. This also makes Settings -> Back -> Start safe.
            if key == 'left mouse down':
                self.move_keys.clear()
                self.mouse_free = True
                self._lock_mouse(False)
            if key == 'escape':
                if mode == 'settings':
                    self._show_menu()
                else:
                    application.quit()
                return
            if mode == 'menu':
                if key == '1':
                    self._enter_safe_zone(from_load=False)
                    return
                if key == '2':
                    self._try_load_game()
                    return
                if key == '3':
                    self._enter_settings()
                    return
                return
            # settings
            if key == '1':
                # mouse sens — bump through presets
                presets = (80.0, 110.0, 140.0, 180.0, 220.0)
                try:
                    i = presets.index(float(self.mouse_sens))
                    self.mouse_sens = presets[(i + 1) % len(presets)]
                except ValueError:
                    self.mouse_sens = 140.0
                self._refresh_menu_labels()
                self.toast('Mouse sensitivity %d' % int(self.mouse_sens))
                return
            if key == '2':
                self.fullscreen_stub = not bool(getattr(self, 'fullscreen_stub', False))
                try:
                    self.window.fullscreen = self.fullscreen_stub
                except Exception:
                    pass
                self._refresh_menu_labels()
                self.toast('Fullscreen stub ' + ('ON' if self.fullscreen_stub else 'OFF'))
                return
            if key == '3':
                compact = getattr(self, 'render_range', 'full') == 'full'
                self.render_range = 'compact' if compact else 'full'
                opts = getattr(self, 'render_opts', None) or {}
                base_ped = 42.0 if opts.get('quality') != 'low' else 32.0
                base_traffic = 55.0 if opts.get('quality') != 'low' else 40.0
                opts['cull_ped'] = base_ped * (0.70 if compact else 1.0)
                opts['cull_traffic'] = base_traffic * (0.70 if compact else 1.0)
                self.render_opts = opts
                self._refresh_menu_labels()
                self.toast('Compact render range ' + ('ON' if compact else 'OFF'))
                return
            if key == '4':
                self._show_menu()
                return
            return

        # Safe yard: Esc always returns to title (never soft-lock).
        if mode == 'safe' and key == 'escape':
            self._show_menu()
            self.toast('Title menu.')
            return

        if key in ('enter', 'return') and getattr(self, 'tutorial_active', False) and mode == 'play':
            self._finish_tutorial()
            return
        if key == 'escape':
            if self.ui_open:
                self.close_panel()
                return
            if mode == 'safe':
                self._show_menu()
                return
            application.quit()
            return
        if self.ui_open:
            if key in ('1', '2', '3', '4'):
                self._pick_choice(int(key) - 1)
            return
        if key in ('i', 'I'):
            self.open_inventory()
            return
        if key == 'e':
            if mode == 'safe':
                if self.in_car:
                    self._exit_car()
                    return
                self._try_safe_gate()
                return
            self._interact()
            return
        if mode != 'play':
            # Safe yard: movement remains active with or without mouse look.
            if key == 'left mouse down':
                if not getattr(self.mouse, 'locked', False):
                    self._lock_mouse(True)
                    self.mouse_free = False
                    self.mouse_look_cooldown = 2
                    self.toast('Mouse locked. WASD walk. E at pink gate.')
                return
            if key == 'tab':
                self.move_keys.clear()
                self.mouse_free = not getattr(self, 'mouse_free', False)
                self._lock_mouse(not self.mouse_free)
                self.mouse_look_cooldown = 2
                self.toast('Cursor free (Tab to recapture)' if self.mouse_free else 'Mouse locked. WASD to walk.')
            return
        if key == '1':
            self._toggle_pistol()
            return
        if key in ('h', 'H'):
            self._eat_food()
            return
        if key == 'left mouse down':
            # Clicking the window should capture the mouse for look+WASD.
            if not self.ui_open and not getattr(self.mouse, 'locked', False):
                self._lock_mouse(True)
                self.mouse_free = False
                self.mouse_look_cooldown = 2
                self.toast('Mouse locked. WASD walk. E talk. Tab frees cursor.')
                return
            if self.pistol_drawn:
                self._shoot()
            return
        if key == 'tab':
            self.mouse_free = not getattr(self, 'mouse_free', False)
            self._lock_mouse(not self.mouse_free)
            self.mouse_look_cooldown = 2
            self.toast('Cursor free (Tab to recapture)' if self.mouse_free else 'Mouse locked. WASD to walk.')

    def _try_safe_gate(self):
        gate = getattr(self, 'safe_gate', None)
        if not gate or not self.player:
            self.toast('No gate.')
            return
        pos = self._actor_pos()
        d = dist_xz(pos, (gate.x, 0, gate.z))
        if d <= 3.2:
            self._leave_safe_to_world()
            return
        # Test pickup is intentionally local to the safe yard and does not
        # alter normal city progression unless the player chooses it.
        for pickup in list(getattr(self, 'safe_debug_pickups', []) or []):
            ent = pickup.get('entity') if isinstance(pickup, dict) else None
            if ent and dist_xz(pos, (ent.x, 0, ent.z)) <= 2.2:
                if pickup.get('kind') == 'pistol':
                    self.owned_pistol = True
                    self.ammo = max(int(getattr(self, 'ammo', 0)), 60)
                    try:
                        ent.enabled = False
                    except Exception:
                        pass
                    self.safe_debug_pickups.remove(pickup)
                    self.toast('Debug pistol acquired — 60 rounds. Press 1 to draw.')
                    return
        car, cd = self._nearest_car(pos, 3.0)
        if car and getattr(car, 'safe_yard', False):
            self._enter_car(car)
            return
        self.toast('Safe yard: E at DEBUG PISTOL, the blue test car, or pink gate.')

    def _toggle_pistol(self):
        if not self.owned_pistol:
            self.toast('No pistol. Buy one from Gage at Hancock Gun Hut (east).')
            return
        self.pistol_drawn = not self.pistol_drawn
        if self.gun_model and not self.in_car:
            self.gun_model.enabled = self.pistol_drawn
        if self.pistol_drawn:
            self.panic_t = max(self.panic_t, 3.5)
            self.toast('Pistol drawn.')
        else:
            self.toast('Holstered.')

    def _interact(self):
        if getattr(self, 'mode', 'play') in ('menu', 'settings'):
            return
        pos = self._actor_pos()
        if getattr(self, 'mode', '') == 'safe':
            if getattr(self, 'safe_gate', None):
                g = self.safe_gate
                if dist_xz(pos, (g.x, 0, g.z)) < 3.5:
                    self._leave_safe_to_world()
                    return
            self.toast('Walk to the pink gate (west) — E to start Clermont.')
            return
        if getattr(self, 'active_interior', None):
            self._leave_interior()
            return
        gate, gate_d = self._nearest_region_gate(pos)
        if gate:
            self._travel_region_gate(gate)
            return
        gas = self._nearest_gas(pos)
        if self.in_car:
            if gas:
                self.open_gas_station(gas, self.in_car)
                return
            self._exit_car()
            return
        if gas:
            near_car, _ncd = self._nearest_car(pos, 5.0)
            # Prefer parked/player car near pump; traffic ignored for service
            service = None
            if near_car and not getattr(near_car, 'traffic', False):
                service = near_car
            if service is not None:
                self.open_gas_station(gas, service)
                return
            # A pedestrian enters the actual market instead of being blocked
            # by the vehicle-service-only pump interaction.
            npc_chk, nd_chk = self._nearest_npc(pos, 3.0)
            if not (npc_chk and nd_chk <= 3.0):
                self._enter_interior('gas_market')
                return
        car, cd = self._nearest_car(pos, 3.2)
        npc, nd = self._nearest_npc(pos, 3.0)
        stall, sd = self._nearest_stall(pos, 3.6)

        # prefer NPC / stall over car if closer
        if npc and nd <= 3.0 and (car is None or nd <= cd + 0.15):
            kind = getattr(npc, 'kind', '')
            if kind == 'mira':
                self.open_mira('start')
                return
            if kind == 'michelle':
                nid = 'door' if not self.michelle_seen else (self.michelle_node or 'door')
                self.michelle_seen = True
                self.michelle_auto_opened = True
                self.open_michelle(nid)
                self._quest_event('talk', npc_id='michelle')
                self._tutorial_note('talk')
                return
            if kind == 'gun':
                self.open_gun_shop()
                return
